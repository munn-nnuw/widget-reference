【AI用プロンプト】
固定イントロ以外、次に選べる具体的な行動があるときだけ、本文の末尾に <DraftChoices choices={["周囲を調べる", "店主に尋ねる"]} /> の形のタグを一つコード枠外に出す。choices は状況に合う短い行動文1〜3件。重複させない。例文を毎回使わない。候補表示・下書きを行動の成立として扱わない。利用者が実際に送信した最新の文を読み、その場の状況で結果を描く。

【プレビュー用タグ】
<DraftChoices choices={["周囲を調べる", "店主に尋ねる"]} />

【コード】
function DraftChoices({ choices }) {
  const usable = Array.isArray(choices)
    ? choices.filter(value => typeof value === 'string').map(value => value.trim()).filter(Boolean)
    : [];
  const options = [...new Set(usable)].slice(0, 3);
  const [state, setState] = useState('ready');
  const busy = useRef(false);

  async function choose(text) {
    if (busy.current) return;
    busy.current = true;
    setState('pending');
    try {
      const result = await pinx.chat.insertText(text);
      if (result?.ok) {
        setState('done');
        return;
      }
    } catch {
      // 入力欄が使えない場合も下の表示を使う。
    }
    busy.current = false;
    setState('error');
  }

  if (!options.length) return <p>現在の行動候補はありません。</p>;

  return (
    <section aria-label="行動候補">
      <ul>
        {options.map(option => (
          <li key={option}>
            <button type="button" onClick={() => choose(option)} disabled={state === 'pending' || state === 'done'}>
              {option}
            </button>
          </li>
        ))}
      </ul>
      <p role="status">
        {state === 'done' && '下書きに入りました。確認して送信してください。'}
        {state === 'error' && '入力欄を確認して、もう一度試してください。'}
      </p>
    </section>
  );
}
