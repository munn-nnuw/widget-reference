【AI用プロンプト】
固定イントロ以外の各AI返答末尾に <TaskList tasks={[{id:"repair",kind:"quest",title:"時計の修理",next:"部品を探す"}]} /> の形のタグを一つコード枠外に出す。tasks は実際に引き受けた未完の全件。id は同じ用件の固定文字列、kind は quest（依頼）か promise（約束）、title は件名、next は確定している次の行動か空文字。新規開始は空、途中は直近一覧を継ぐ。提案・言及は未登録。成立した変更だけ更新し、完了・取消が確定した項目だけ外す。確定した空は tasks={[]}、一覧が不明なら属性を省略。例を事実化しない。

【プレビュー用タグ】
<TaskList tasks={[{id:"repair",kind:"quest",title:"時計の修理",next:"部品を探す"}]} />

【コード】
function TaskList({ tasks }) {
  const valid = Array.isArray(tasks) && tasks.every(task =>
    task && typeof task.id === 'string' && task.id.trim() &&
    (task.kind === 'quest' || task.kind === 'promise') &&
    typeof task.title === 'string' && task.title.trim() && typeof task.next === 'string'
  ) && new Set(tasks.map(task => task.id)).size === tasks.length;

  if (!valid) return <p>用件は不明です。</p>;
  if (!tasks.length) return <p>引き受けている用件はありません。</p>;

  return (
    <ul>
      {tasks.map(task => (
        <li key={task.id}>
          {task.kind === 'quest' ? '依頼' : '約束'} · {task.title}
          {task.next.trim() && <div>次にやること {task.next}</div>}
        </li>
      ))}
    </ul>
  );
}
