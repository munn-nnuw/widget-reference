【AI用プロンプト】
固定イントロ以外の各AI返答末尾に <WalletLedger before={10} entries={[{id:"trade-1",delta:-3}]} /> の形のタグを一つコード枠外に出す。例は事実化しない。before は今回の取引前残高。新規開始時の初期値は0、以後は直近の確定結果を継ぐ。entries は今回の本文で受け渡しが成立した取引だけ。受取は正、支払いは負の整数。同じ取引には同じidを使い、済んだ取引は次のタグへ再送しない。提示、未払い、予定、引用の金額は含めない。収支なしは entries={[]}。残高や取引が確定できないときは当該属性を省き、0を推測しない。

【プレビュー用タグ】
<WalletLedger before={10} entries={[{id:"trade-1",delta:-3}]} />

【コード】
function WalletLedger({ before, entries }) {
  if (!Number.isSafeInteger(before) || before < 0 || !Array.isArray(entries)) {
    return <p>所持金は不明です。</p>;
  }

  const transactions = new Map();
  for (const entry of entries) {
    if (!entry || typeof entry.id !== 'string' || !entry.id.trim() ||
        !Number.isSafeInteger(entry.delta) || !entry.delta) {
      return <p>所持金は不明です。</p>;
    }
    if (transactions.has(entry.id) && transactions.get(entry.id) !== entry.delta) {
      return <p>同じ取引の金額が一致しません。</p>;
    }
    transactions.set(entry.id, entry.delta);
  }

  let balance = before;
  for (const delta of transactions.values()) {
    balance += delta;
    if (!Number.isSafeInteger(balance)) {
      return <p>今回の取引後の所持金を確定できません。</p>;
    }
  }
  if (balance < 0) {
    return <p>今回の取引後の所持金を確定できません。</p>;
  }

  return <p>所持金 {balance} 枚</p>;
}
